// SCRATCH (design only, not for commit): puts a lobby on screen without a room.
// ?fake=host|guest|local [&n=players] [&mode=turns|race|delve] [&diff=...]
import { session } from './lib/session.svelte';
import { createGame, DEFAULT_SETTINGS, type GameMode, type Difficulty } from './lib/game';

const LONG = ['Kalguur', 'Doryani the Wise One', 'Una of the Vaal Ruins', 'Rog', 'Sekhema Asala Ascended'];
const NAMES = ['Kalguur', 'Doryani', 'Una of the Vaal', 'Rog', 'Alva', 'Dannig', 'Hooded One', 'Zelina', 'Oswin', 'Finn', 'Sekhema Asala', 'Renly'];

export function fakeLobby(p: URLSearchParams) {
  const kind = p.get('fake');
  if (!kind) return;
  const n = Number(p.get('n') ?? 5);
  const mode = (p.get('mode') ?? 'turns') as GameMode;
  const settings = { ...DEFAULT_SETTINGS, mode, ...(p.get('diff') ? { difficulty: p.get('diff') as Difficulty } : {}), ...(p.has('pub') ? { public: true } : {}), ...(p.has('pts') ? { targetScore: Number(p.get('pts')) } : {}) };
  const ids = NAMES.slice(0, n).map((_, i) => `p${i}`);
  session.startLocal(createGame(kind === 'local' ? null : ids[0], settings));
  session.hideCode = p.has('hide');
  const diff = p.get('diff');
  if (kind !== 'local') {
    session.mode = kind === 'guest' ? 'client' : 'host';
    session.code = p.get('code') ?? 'KXR4QT';
    session.myPlayerId = kind === 'guest' ? ids[2] : ids[0];
  }
  ids.forEach((id, i) => session.labAct({ type: 'join', playerId: id, name: (p.has('long') ? LONG : NAMES)[i] ?? NAMES[i] }, kind === 'local' ? null : id));
  if (p.has('locked')) session.labAct({ type: 'settings', settings: { locked: true } }, kind === 'local' ? null : ids[0]);
}
